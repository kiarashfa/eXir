<Step id="fill">
  Fill a Collins glass with ice.
</Step>

<Step id="pour">
  Pour in the <Qty ref="ginger-beer"/>.
</Step>

<Step id="float">
  Pour the <Qty ref="rum"/> slowly over the back of a bar spoon so it sits on
  top rather than mixing in, and leave it unstirred.
</Step>

<Step id="garnish">
  Set a lime wedge on the rim.
</Step>
