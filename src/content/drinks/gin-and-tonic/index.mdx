<Step id="fill">
  Fill a tall Collins glass with ice.
</Step>

<Step id="pour">
  Pour in the <Qty ref="gin"/>.
</Step>

<Step id="top">
  Top with the <Qty ref="tonic"/> and lift once with a bar spoon.
</Step>

<Step id="garnish">
  Squeeze a lime wedge over the drink and drop it in.
</Step>
