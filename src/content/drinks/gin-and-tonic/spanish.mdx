<Step id="chill">
  Chill the glass.
</Step>

<Step id="fill">
  Fill the glass to the rim with ice.
</Step>

<Step id="pour">
  Pour in the <Qty ref="gin"/>.
</Step>

<Step id="garnish">
  Cut a wide strip of lemon peel, express its oil over the drink and drop it
  in.
</Step>

<Step id="top">
  Top slowly with the <Qty ref="tonic"/>, pouring down the inside of the
  glass to keep the bead, and lift once with a bar spoon.
</Step>
