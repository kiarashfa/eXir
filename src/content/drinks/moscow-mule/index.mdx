<Step id="fill">
  Fill the copper mug with cubed ice.
</Step>

<Step id="pour">
  Pour in the <Qty ref="vodka"/> and the <Qty ref="lime"/>.
</Step>

<Step id="top">
  Top with the <Qty ref="ginger-beer"/>, poured gently down the side of the
  mug rather than straight onto the ice.
</Step>

<Step id="garnish">
  Drop in a lime wheel.
</Step>
