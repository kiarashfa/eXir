<Step id="warm">
  Combine the <Qty ref="wine"/>, the <Qty ref="sugar"/>, the
  <Qty ref="cinnamon"/>, the <Qty ref="cloves"/>, the <Qty ref="star-anise"/>
  and the <Qty ref="orange"/> in a pot and warm gently until the sugar
  dissolves and the wine steams.
</Step>

<Step id="steep">
  Take off the heat, cover and let the spices steep for <Dur step="steep"/>.
</Step>

<Step id="pour">
  Warm through again briefly and strain hot into mugs.
</Step>
