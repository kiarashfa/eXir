<Step id="boil">
  Combine the <Qty ref="corn"/>, <Qty ref="pineapple"/>, <Qty ref="cinnamon"/>
  and <Qty ref="cloves"/> with the <Qty ref="water"/> and boil at
  <Temp c={100}/> for <Dur step="boil"/>, until the liquid turns a deep
  purple.
</Step>

<Step id="strain">
  Strain out the corn, pineapple and spices and discard them.
</Step>

<Step id="sweeten">
  Stir the <Qty ref="sugar"/> into the hot strained liquid until dissolved.
</Step>

<Step id="chill">
  Cool for <Dur step="chill"/>, then stir in the <Qty ref="lime-juice"/> and
  pour over ice.
</Step>
