<Step id="fill">
  Fill a large wine glass with ice.
</Step>

<Step id="pour">
  Pour the <Qty ref="aperol"/> over the ice.
</Step>

<Step id="top">
  Top with the <Qty ref="prosecco"/> and then the <Qty ref="soda"/>, and
  give the drink one gentle lift with a bar spoon rather than a stir.
</Step>

<Step id="garnish">
  Drop in a half-wheel of orange.
</Step>
