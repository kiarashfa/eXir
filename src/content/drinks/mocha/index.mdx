<Step id="grind">
  Grind the <Qty ref="coffee"/> fine for the espresso machine.
</Step>

<Step id="extract">
  Pump the <Qty ref="water"/> through the grounds at <Temp c={93}/> for
  <Dur step="extract"/>, into a glass large enough to take the milk after.
</Step>

<Step id="paste">
  Stir the <Qty ref="cocoa"/> and <Qty ref="sugar"/> into the hot shot for
  <Dur step="paste"/>, until it forms a smooth, glossy paste with no dry
  lumps left.
</Step>

<Step id="steam">
  Steam the <Qty ref="milk"/> to a light microfoam.
</Step>

<Step id="pour">
  Pour the steamed milk over the paste, finishing with a thin cap of foam.
</Step>
