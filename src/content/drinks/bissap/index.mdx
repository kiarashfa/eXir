<Step id="steep">
  Pour the <Qty ref="water"/> at <Temp c={100}/> over the <Qty ref="hibiscus"/>
  and the <Qty ref="mint"/>, and leave to steep for <Dur step="steep"/>.
</Step>

<Step id="strain">
  Strain out the calyces and mint and discard them.
</Step>

<Step id="sweeten">
  Stir in the <Qty ref="sugar"/> while the liquid is still hot enough to
  dissolve it, then add the <Qty ref="flower-water"/>.
</Step>

<Step id="pour">
  Pour over cubed ice in the glass and serve at once.
</Step>
