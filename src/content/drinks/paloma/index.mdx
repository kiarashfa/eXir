<Step id="rim">
  Run a lime wedge around half the rim of a highball glass and dip that
  half in salt.
</Step>

<Step id="fill">
  Fill the glass with cubed ice.
</Step>

<Step id="pour">
  Pour in the <Qty ref="tequila"/> and the <Qty ref="lime"/>.
</Step>

<Step id="top">
  Top with the <Qty ref="soda"/> and set a lime wedge on the rim.
</Step>
