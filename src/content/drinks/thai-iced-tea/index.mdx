<Step id="steep">
  Pour the <Qty ref="water"/> at <Temp c={100}/> over the <Qty ref="tea"/>
  and the <Qty ref="anise"/> and steep for <Dur step="steep"/>.
</Step>

<Step id="strain">
  Strain out the leaf and spice.
</Step>

<Step id="sweeten">
  Stir in the <Qty ref="condensed"/> while the tea is still hot enough to
  dissolve it fully.
</Step>

<Step id="pour">
  Let cool, then pour over cubed ice in a tall glass.
</Step>

<Step id="float">
  Pour the <Qty ref="evap"/> slowly over the back of a spoon so it sits on
  top as a pale layer.
</Step>
