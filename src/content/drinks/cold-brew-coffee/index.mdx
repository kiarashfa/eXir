<Step id="grind">
  Grind the <Qty ref="coffee"/> coarse, coarser than a drip grind.
</Step>

<Step id="build">
  Combine the grounds with the <Qty ref="water"/> in the jar and stir to wet
  them evenly.
</Step>

<Step id="steep">
  Seal the jar and leave it at room temperature or in the refrigerator for
  <Dur step="steep"/>.
</Step>

<Step id="strain">
  Strain out the grounds through a paper filter or fine cloth and pour over
  ice.
</Step>
