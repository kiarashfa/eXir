<Step id="steep">
  Pour the <Qty ref="water"/> at <Temp c={100}/> over the <Qty ref="tea"/> and
  leave to steep for <Dur step="steep"/>.
</Step>

<Step id="strain">
  Strain out the leaf and discard it.
</Step>

<Step id="sweeten">
  Stir in the <Qty ref="sugar"/> while the tea is still hot, until fully
  dissolved.
</Step>

<Step id="pour">
  Pour over cubed ice in the glass and serve at once.
</Step>
