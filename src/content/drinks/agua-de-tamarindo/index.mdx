<Step id="steep">
  Pour the <Qty ref="water"/> at <Temp c={100}/> over the <Qty ref="tamarind"/>
  and leave to soften for <Dur step="steep"/>.
</Step>

<Step id="mash">
  Mash and work the softened pulp through a sieve set over a jug for
  <Dur step="mash"/>, pressing the flesh through and leaving the fibres and
  seeds behind.
</Step>

<Step id="strain">
  Stir the <Qty ref="sugar"/> into the strained liquid until fully dissolved.
</Step>

<Step id="pour">
  Pour over cubed ice in the glass and serve at once.
</Step>
