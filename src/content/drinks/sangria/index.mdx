<Step id="build">
  Combine the <Qty ref="wine"/>, the <Qty ref="brandy"/>, the
  <Qty ref="syrup"/>, the <Qty ref="orange"/> and the <Qty ref="lemon"/> in
  a pitcher.
</Step>

<Step id="macerate">
  Refrigerate for <Dur step="macerate"/>, stirring once or twice.
</Step>

<Step id="pour">
  Pour over cubed ice into red wine glasses, dividing the fruit between
  them.
</Step>

<Step id="top">
  Top each glass with the <Qty ref="soda"/>.
</Step>
