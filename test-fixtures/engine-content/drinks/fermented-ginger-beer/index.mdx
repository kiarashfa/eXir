<Step id="grate">
  Grate the <Qty ref="ginger"/> coarsely, skin and all, over <Dur step="grate"/>.
</Step>

<Step id="combine">
  Stir the <Qty ref="sugar"/> into the <Qty ref="water"/> until it has gone,
  then add the ginger and the <Qty ref="lime"/>.
</Step>

<Step id="primary">
  Cover loosely and leave at room temperature, stirring once a day, until it
  smells yeasty and tastes dry rather than sweet.
</Step>

<Step id="bottle">
  Strain and bottle, leaving a finger of headroom in each.
</Step>

<Step id="secondary">
  Seal and leave to carbonate, venting once a day.
</Step>
